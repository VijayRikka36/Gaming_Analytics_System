create database gaming_analytics;

use  gaming_analytics;

create table gamers(
gamer_id int primary key,
gamer_name varchar(50) NOT NULL,
gamer_username varchar(50) unique not null,
gamer_email varchar(50) unique,
country varchar(50),
registration_date Date not null
);
desc gamers;
 select*from gamers;
 
 
 
 create table games(
 gamer_id int primary key,
 game_title varchar(50) not null,
 developer_name varchar(100) not null,
 release_date date,
 price decimal(10,2),
 age_rate varchar(10)
 );
 desc games;
 select*from games;
 
 create table genres(
 genre_id int primary key,
 genre_name varchar(50) unique not null,
 genre_description varchar(200)
 );
 desc genres;
 select*from genres;
 
 
 create table platforms(
 platform_id int primary key,
 platform_name varchar(50) unique not null,
 manufacturer varchar(100)
 );
 desc platforms;
 select*from platforms;
 
 
create table game_reviews(
review_id int primary key,
gamer_id int,
game_id int,
rating int,
review_text varchar(500),
review_date date not null,
foreign key(gamer_id) references gamers(gamer_id),
foreign key(game_id) references games(game_id)
);
desc game_reviews;
select*from game_reviews;
alter table games rename column  gamer_id to game_id;



create table game_genres(
game_id int,
genre_id int,
primary key(game_id,genre_id),
foreign key(game_id) references games(game_id),
foreign key(genre_id) references genres(genre_id)
);
desc game_genres;
select*from game_genres;


create table game_platforms(
game_id int,
platform_id int,
primary key(game_id, platform_id),
foreign key(game_id) references games(game_id),
foreign key(platform_id) references platforms(platform_id)
);
desc game_platforms;
select*from game_platforms;


create table achievements(
achievement_id int primary key,
gamer_id int,
game_id int,
achievement_name varchar(200) not null,
foreign key(gamer_id) references gamers(gamer_id),
foreign key(game_id) references games(game_id)
);
desc achievements;
select*from achievements;

CREATE OR REPLACE VIEW view_game_details AS
SELECT
    game_id,
    game_title,
    developer_name,
    release_date,
    price,
    age_rate
FROM games;

select*from view_game_details;

CREATE OR REPLACE VIEW view_gamer_reviews AS
SELECT
    gm.gamer_id,
    gm.gamer_name,
    g.game_title,
    r.rating,
    r.review_text,
    r.review_date
FROM game_reviews r
JOIN gamers gm
    ON r.gamer_id = gm.gamer_id
JOIN games g
    ON r.game_id = g.game_id;
    
    SELECT * FROM view_gamer_reviews;
    
CREATE OR REPLACE VIEW view_game_rating_summary AS
SELECT
    g.game_id,
    g.game_title,
    COUNT(r.review_id) AS total_reviews,
    ROUND(AVG(r.rating), 2) AS avg_rating
FROM games g
LEFT JOIN game_reviews r
    ON g.game_id = r.game_id
GROUP BY g.game_id, g.game_title;

SELECT * FROM view_game_rating_summary;

CREATE OR REPLACE VIEW view_gamer_achievement_summary AS
SELECT
    gm.gamer_id,
    gm.gamer_name,
    COUNT(a.achievement_id) AS total_achievements
FROM gamers gm
LEFT JOIN achievements a
    ON gm.gamer_id = a.gamer_id
GROUP BY gm.gamer_id, gm.gamer_name;

SELECT * FROM view_gamer_achievement_summary;

DROP PROCEDURE IF EXISTS sp_games_by_genre;
DELIMITER $$
CREATE PROCEDURE sp_games_by_genre(IN p_genre_id INT)
BEGIN
    SELECT g.game_id, g.game_title, g.developer_name,
           g.release_date, g.price, g.age_rating
    FROM games g
    JOIN game_genres gg ON g.game_id = gg.game_id
    WHERE gg.genre_id = p_genre_id
    ORDER BY g.release_date DESC;
END $$
DELIMITER ;


DROP PROCEDURE IF EXISTS sp_game_rating_summary;
DELIMITER $$
CREATE PROCEDURE sp_game_rating_summary(IN p_game_id INT)
BEGIN
    SELECT g.game_title,
           COUNT(r.review_id) AS total_reviews,
           ROUND(AVG(r.rating), 2) AS avg_rating
    FROM games g
    LEFT JOIN game_reviews r ON g.game_id = r.game_id
    WHERE g.game_id = p_game_id
    GROUP BY g.game_id, g.game_title;
END $$
DELIMITER ;


DROP PROCEDURE IF EXISTS sp_add_game_review;
DELIMITER $$
CREATE PROCEDURE sp_add_game_review(
    IN p_gamer_id INT,
    IN p_game_id INT,
    IN p_rating INT,
    IN p_review_text VARCHAR(500),
    IN p_review_date DATE
)
BEGIN
    IF p_rating < 1 OR p_rating > 10 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Rating must be between 1 and 10';
    ELSE
        INSERT INTO game_reviews
            (gamer_id, game_id, rating, review_text, review_date)
        VALUES
            (p_gamer_id, p_game_id, p_rating, p_review_text, p_review_date);
    END IF;
END $$
DELIMITER ;


DROP PROCEDURE IF EXISTS sp_achievements_by_gamer;
DELIMITER $$
CREATE PROCEDURE sp_achievements_by_gamer(IN p_gamer_id INT)
BEGIN
    SELECT gm.gamer_name, g.game_title, a.achievement_name
    FROM achievements a
    JOIN gamers gm ON a.gamer_id = gm.gamer_id
    JOIN games g ON a.game_id = g.game_id
    WHERE a.gamer_id = p_gamer_id
    ORDER BY g.game_title;
END $$
DELIMITER ;

DROP TRIGGER IF EXISTS trg_validate_review_rating;
DELIMITER $$
CREATE TRIGGER trg_validate_review_rating
BEFORE INSERT ON game_reviews
FOR EACH ROW
BEGIN
    IF NEW.rating < 1 OR NEW.rating > 10 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Rating must be between 1 and 10';
    END IF;
END $$
DELIMITER ;


DROP TRIGGER IF EXISTS trg_validate_game_price;
DELIMITER $$
CREATE TRIGGER trg_validate_game_price
BEFORE INSERT ON games
FOR EACH ROW
BEGIN
    IF NEW.price < 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Game price cannot be negative';
    END IF;
END $$
DELIMITER ;



DROP TRIGGER IF EXISTS trg_validate_game_price_update;
DELIMITER $$
CREATE TRIGGER trg_validate_game_price_update
BEFORE UPDATE ON games
FOR EACH ROW
BEGIN
    IF NEW.price < 0 THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Game price cannot be negative';
    END IF;
END $$
DELIMITER ;


DROP TRIGGER IF EXISTS trg_prevent_gamer_delete;
DELIMITER $$
CREATE TRIGGER trg_prevent_gamer_delete
BEFORE DELETE ON gamers
FOR EACH ROW
BEGIN
    IF EXISTS (
        SELECT 1
        FROM game_reviews
        WHERE gamer_id = OLD.gamer_id
    ) THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT =
            'Cannot delete gamer who has submitted reviews';
    END IF;
END $$
DELIMITER ;
